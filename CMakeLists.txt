# Generate JSON with CMake's JSON encoder rather than interpolating user paths
# or allowlists into shell commands. VERBATIM execute_process uses no shell.
function(ff_json_string output value)
  string(REPLACE "\\" "\\\\" value "${value}")
  string(REPLACE "\"" "\\\"" value "${value}")
  string(REPLACE "\n" "\\n" value "${value}")
  string(REPLACE "\r" "\\r" value "${value}")
  set(${output} "\"${value}\"" PARENT_SCOPE)
endfunction()
set(ff_options "{}")
foreach(key profile disable)
  string(TOUPPER "${key}" upper)
  ff_json_string(value "${FFMPEG_${upper}}")
  string(JSON ff_options SET "${ff_options}" "${key}" "${value}")
endforeach()
foreach(pair "arch|${FF_ARCH}" "compiler|${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}" "system|${CMAKE_SYSTEM_NAME}" "shared_suffix|${CMAKE_SHARED_LIBRARY_SUFFIX}" "datadir|${CMAKE_INSTALL_FULL_DATADIR}/ffmpeg")
  string(REPLACE "|" ";" pair "${pair}")
  list(GET pair 0 key)
  list(GET pair 1 value)
  ff_json_string(value "${value}")
  string(JSON ff_options SET "${ff_options}" "${key}" "${value}")
endforeach()
if(FFMPEG_PROGRAMS)
  string(JSON ff_options SET "${ff_options}" programs true)

  string(JSON ff_options SET "${ff_options}" programs false)
endif()
string(JSON ff_options SET "${ff_options}" components "{}")

  ff_json_string(value "${FFMPEG_${upper}}")
  string(JSON ff_options SET "${ff_options}" components "${kind}" "${value}")

foreach(lib IN LISTS FF_ENABLED_LIBS)
  foreach(asm_source IN LISTS FF_ASM_${lib})
    get_filename_component(asm_dir "${asm_source}" DIRECTORY)
    set_source_files_properties("${asm_source}" PROPERTIES COMPILE_OPTIONS "-I${asm_dir}/")
  endforeach()
  set(compile_target ${lib})
  if(WIN32 AND BUILD_SHARED_LIBS)
    set(compile_target ${lib}_objects)
    add_library(${compile_target} OBJECT ${FF_SOURCES_${lib}} ${FF_ASM_${lib}})
    set(objects "${FF_GENERATED}/${lib}-$<CONFIG>.objects.txt")
    set(exports "${FF_GENERATED}/${lib}-$<CONFIG>.def")
    file(GENERATE OUTPUT "${objects}" CONTENT "$<JOIN:$<TARGET_OBJECTS:${compile_target}>,\n>\n")
    add_custom_command(OUTPUT "${exports}"
      COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tools/exports.py"
        --cmake "${CMAKE_COMMAND}" --objects "${objects}"
        --version-script "${FFMPEG_SOURCE_DIR}/lib${lib}/lib${lib}.v" --output "${exports}"
      DEPENDS ${compile_target} $<TARGET_OBJECTS:${compile_target}> "${objects}"
        "${FFMPEG_SOURCE_DIR}/lib${lib}/lib${lib}.v" "${PROJECT_SOURCE_DIR}/tools/exports.py" VERBATIM)
    add_library(${lib} SHARED $<TARGET_OBJECTS:${compile_target}> "${exports}")
    target_link_libraries(${compile_target} PRIVATE ffmpeg_build_options ${FF_EXTERNAL_LIBS})
  else()
    add_library(${lib} ${FF_SOURCES_${lib}} ${FF_ASM_${lib}})
  endif()
  set_target_properties(${lib} PROPERTIES VERSION "${FF_MAJOR_${lib}}" SOVERSION "${FF_MAJOR_${lib}}")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND BUILD_SHARED_LIBS)
    target_link_options(${lib} PRIVATE "LINKER:--version-script=${FF_GENERATED}/lib${lib}.ver")
    set_property(TARGET ${lib} APPEND PROPERTY LINK_DEPENDS "${FF_GENERATED}/lib${lib}.ver")
  endif()
  target_compile_definitions(${compile_target} PRIVATE $<$<COMPILE_LANGUAGE:C,CXX,OBJC>:HAVE_AV_CONFIG_H>)
  if(lib STREQUAL "swscale" AND TARGET ffmpeg_uops)
    add_dependencies(${compile_target} ffmpeg_uops)
    target_compile_options(${compile_target} PRIVATE "$<$<COMPILE_LANGUAGE:ASM_NASM>:-I${FF_GENERATED}/libswscale/x86/>")
  endif()
  add_library(FFmpeg::${lib} ALIAS ${lib})
  target_link_libraries(${lib} PRIVATE "$<BUILD_INTERFACE:ffmpeg_build_options>" ${FF_EXTERNAL_LIBS})
  target_include_directories(${lib} PUBLIC "$<BUILD_INTERFACE:${FF_GENERATED}>" "$<BUILD_INTERFACE:${FFMPEG_SOURCE_DIR}>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
  if(WIN32)
    target_link_libraries(${lib} PRIVATE ws2_32 bcrypt crypt32 ncrypt ole32 oleaut32 user32 psapi shell32 shlwapi secur32 strmiids uuid dxguid gdi32 vfw32 mfplat mfuuid)
  else()
    target_link_libraries(${lib} PRIVATE Threads::Threads m ${CMAKE_DL_LIBS})
  endif()
  install(TARGETS ${lib} EXPORT FFmpegTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(FILES ${FF_HEADERS_${lib}} DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}/lib${lib}")
endforeach()
